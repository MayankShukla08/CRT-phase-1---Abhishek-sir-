// Problem -- 1 Stack

#include <iostream>
#include <stack>
using namespace std;

int main()
{
    int arr[] = {10,20,30,40,50};
    int n = 5;
    
    stack<int> st;
    //java -- Stack<integer> st = new Stack<>();
    
    for (int i = 0; i < n; i++){
        st.push(arr[i]);
    }
    // pop the data from the stack and put into the array back
    for (int i = 0; i < n; i++){
        arr[i] = st.top();
        st.pop();
    }
    
    cout <<"Reversed array.."<<"\n";
    
    for (int i = 0; i < n; i++){
        cout << arr[i]<< " ";
    }
    
    return 0;
}
